# ESP32 INVERTER OTA RELEASE PROCEDURE
# Repository: Jezzylinks/inverter
# Platform: Windows PowerShell / VS Code
# Project: Inverter_System_FileSystem

cd "C:\Users\hp\Documents\PlatformIO\Projects\Inverter_System_FileSystem"

git status

git branch --show-current

git pull --ff-only origin master

gh auth status

$VERSION = "1.5.1"
$TAG = "v$VERSION"

# The firmware must report the version it is released as. If this header is not
# updated before pio run, a device updated to this release still sees the
# manifest as newer than itself and keeps offering the same update.
(Get-Content include\system\firmware_version.h) -replace '#define INVERTER_FIRMWARE_VERSION ".*"', "#define INVERTER_FIRMWARE_VERSION `"v$VERSION`"" | Set-Content include\system\firmware_version.h -Encoding ascii

git add .

git commit -m "Prepare OTA firmware release 1.5.1"

git push origin master

pio run

$BIN = ".pio\build\esp32dev\firmware.bin"

if (!(Test-Path $BIN)) { throw "Firmware file not found" }

$SIZE = (Get-Item $BIN).Length

$SHA = (Get-FileHash $BIN -Algorithm SHA256).Hash.ToLower()

New-Item -ItemType Directory -Force "release\$VERSION" | Out-Null

Copy-Item $BIN "release\$VERSION\inverter-$VERSION.bin" -Force

$URL = "https://github.com/Jezzylinks/inverter/releases/download/$TAG/inverter-$VERSION.bin"

"version,url,sha256,size`n$VERSION,$URL,$SHA,$SIZE" | Set-Content "release\$VERSION\inverter.csv" -Encoding ascii

Get-Content "release\$VERSION\inverter.csv"

gh release create $TAG ".\release\$VERSION\inverter-$VERSION.bin" ".\release\$VERSION\inverter.csv" --title "Inverter Firmware $VERSION" --notes "Supervised OTA firmware release $VERSION"

gh release view $TAG

Start-Process "https://github.com/Jezzylinks/inverter/releases/latest"
